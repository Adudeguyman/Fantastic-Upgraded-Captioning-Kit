# CUDA runtime libraries for the prebuilt CUDA llama.cpp backend.
#
# llama.cpp's CUDA releases dynamically link the CUDA runtime (libcublas,
# libcudart, libnccl, ...) but the release archives don't ship those libraries, so
# a CUDA launch on a machine with no system CUDA fails with, e.g.
#   "error while loading shared libraries: libcublas.so.12"
#
# The app auto-discovers any of these already present in the environment (PyTorch
# and the nvidia-*-cu12 wheels bundle them under site-packages) and puts their
# folders on the loader path. Install this only if a clean env has none.
#
#   pip install -r requirements-cuda.txt
#
# Linux x86_64 only — these wheels have no Windows/macOS builds, and the
# Vulkan/CPU llama.cpp backends don't need any of it. libcuda.so.1 is NOT here:
# that comes from the NVIDIA driver and can't be pip-installed.
nvidia-cublas-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-cuda-runtime-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-cuda-nvrtc-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-cufft-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-curand-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-cusolver-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-cusparse-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-nvjitlink-cu12; sys_platform == "linux" and platform_machine == "x86_64"
nvidia-nccl-cu12; sys_platform == "linux" and platform_machine == "x86_64"
